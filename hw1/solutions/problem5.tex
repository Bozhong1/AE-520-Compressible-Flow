\section*{Problem 5}

\partlabel{(a)}
$h$: hot block; $c$: cold block. Each block has total heat capacity
$C=500~\mathrm{J/K}$.
% L1 notes PDF p.13, Eqs. (21), (24): first law; rigid isolated two-block system.
\begin{align*}
Q_{\mathrm{ext}}=W_{\mathrm{ext}}=0
&\quad\Longrightarrow\quad \Delta E_h+\Delta E_c=0,\\
C(T_f-500)+C(T_f-300)&=0,\\
T_f&=\boxed{400~\mathrm{K}}.
\end{align*}

\partlabel{(b)}
% L2 notes PDF p.3 (printed p.18), Eq. (31): reversible comparison path.
% L2 notes PDF p.2, Eqs. (25)--(26): entropy balance for isolated systems.
Along a reversible auxiliary path,
\[
\delta Q_{\mathrm{rev}}=dE=C\,dT
\quad\Longrightarrow\quad
\Delta S=\int_{T_i}^{T_f}\frac{C\,dT}{T}
=C\ln\frac{T_f}{T_i}.
\]
\begin{align*}
\Delta S_h&=500\ln\frac{400}{500}
=\boxed{-111.57~\mathrm{J/K}},\\
\Delta S_c&=500\ln\frac{400}{300}
=\boxed{+143.84~\mathrm{J/K}},\\
\Delta S_{\mathrm{universe}}
&=\Delta S_h+\Delta S_c
=\boxed{32.27~\mathrm{J/K}}=S_{\mathrm{gen}}>0.
\end{align*}

\partlabel{(c)}
Take heat transferred from the hot block to the cold block as positive.
% First-law energy cancellation alone permits either algebraic sign of delta Q.
\[
dE_h=-\delta Q,\qquad dE_c=\delta Q
\quad\Longrightarrow\quad dE_h+dE_c=0
\qquad\text{for either sign of }\delta Q.
\]
% L2 notes PDF p.2, Eq. (26), with each block approximated as uniform in temperature.
For uniform block temperatures,
\[
dS_{\mathrm{universe}}
=\delta Q\left(\frac{1}{T_c}-\frac{1}{T_h}\right)\ge0,
\qquad T_h>T_c
\quad\Longrightarrow\quad \delta Q\ge0.
\]
The first law conserves energy; the second law selects hot-to-cold heat flow.
