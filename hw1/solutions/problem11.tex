\section*{Problem 11}
Ideal, calorically perfect gases; $1$: methane, $2$: argon, $0$: initial state.
Neglect partition volume/heat capacity and changes in bulk kinetic/potential energy.
$V=\text{constant}$, $Q_{\mathrm{ext}}=W_{\mathrm{ext}}=0$;
$R_u=8.3144626~\mathrm{J/(mol\,K)}$.

\partlabel{(a)}
% L1 notes PDF p.2: molar mass/basis definitions.
% L2 notes PDF p.5 (printed p.20), Eq. (39): cv=(f/2)R.
% Multiply by molar mass for molar heat capacity; C_i is the total sample heat capacity.
\begin{align*}
n_1&=\frac{m_1}{M_1}=\frac{1}{0.0160425}=62.334424~\mathrm{mol},
&n_2&=\frac{1}{0.039948}=25.032542~\mathrm{mol},\\
\bar c_{v,i}&=\frac{f_i}{2}R_u,
&C_i&=n_i\bar c_{v,i}=m_ic_{v,i},\\
C_1&=n_1(3R_u)=1554.831716~\mathrm{J/K},
&C_2&=n_2\left(\frac32R_u\right)=312.198205~\mathrm{J/K}.
\end{align*}
% L1 notes PDF p.13, Eqs. (21), (24), and L3 notes PDF p.1, Eq. (74).
\begin{align*}
\Delta E_1+\Delta E_2&=Q_{\mathrm{ext}}-W_{\mathrm{ext}}=0,\\
C_1(T_{f,a}-300)+C_2(T_{f,a}-600)&=0,\\
T_{f,a}&=\frac{C_1(300)+C_2(600)}{C_1+C_2}
=\boxed{350.164949~\mathrm{K}}.
\end{align*}
% L2 notes PDF p.9 (printed p.24), Eq. (69), multiplied by species mass.
% Fixed species masses convert v ratios to V ratios; Delta S_i is total entropy in J/K.
\[
V_{i,f}=V_{i,0}
\quad\Longrightarrow\quad
\Delta S_{i,a}=C_i\ln\frac{T_{f,a}}{T_{i,0}}
+n_iR_u\underbrace{\ln\frac{V_{i,f}}{V_{i,0}}}_{0}
=C_i\ln\frac{T_{f,a}}{T_{i,0}}.
\]
\begin{align*}
\Delta S_{1,a}&=1554.831716\ln\frac{350.164949}{300}
=\boxed{+240.411~\mathrm{J/K}},\\
\Delta S_{2,a}&=312.198205\ln\frac{350.164949}{600}
=\boxed{-168.127~\mathrm{J/K}},\\
\Delta S_a&=\Delta S_{1,a}+\Delta S_{2,a}
=\boxed{72.2843~\mathrm{J/K}}=S_{\mathrm{gen},a}>0.
\end{align*}
% L2 notes PDF p.2, Eqs. (25)--(26): positive entropy generation => irreversible.
Heat transfer across the temperature difference is irreversible.

\partlabel{(b)}
Same original initial states; ideal gases have $E_i=E_i(T)$.
% Same first law and CPG energy relation as (a); ideal-mixture energies are additive.
\[
C_1(T_{f,b}-300)+C_2(T_{f,b}-600)=0
\quad\Longrightarrow\quad
\boxed{T_{f,b}=T_{f,a}=350.164949~\mathrm{K}}.
\]
% L3 notes PDF p.1, Eq. (72), converted to pV=nRuT.
\begin{align*}
V_{i,0}&=\frac{n_iR_uT_{i,0}}{p_{i,0}},\qquad p_{i,0}=10^5~\mathrm{Pa},\\
V_{1,0}&=1.554831716~\mathrm{m^3},\qquad
V_{2,0}=1.248792821~\mathrm{m^3},\\
V&=V_{1,0}+V_{2,0}=2.803624536~\mathrm{m^3},\\
p_f&=\frac{(n_1+n_2)R_uT_{f,b}}{V}
=90726.52~\mathrm{Pa}
=\boxed{0.907265~\mathrm{bar}}.
\end{align*}
% L2 notes PDF p.9, Eq. (69) + L4 notes PDF p.5 (printed p.46), Eq. (203).
% Each species can occupy the entire final mixture volume.
\[
V_{i,f}=V,\quad T_{f,b}=T_{f,a}
\quad\Longrightarrow\quad
\Delta S_{i,b}=\Delta S_{i,a}+n_iR_u\ln\frac{V}{V_{i,0}}.
\]
\begin{align*}
\Delta S_{1,b}
&=240.410961+(62.334424)(8.3144626)\ln\frac{2.803624536}{1.554831716}
=\boxed{545.959~\mathrm{J/K}},\\
\Delta S_{2,b}
&=-168.126641+(25.032542)(8.3144626)\ln\frac{2.803624536}{1.248792821}
=\boxed{+0.197252~\mathrm{J/K}},\\
\Delta S_b&=\Delta S_{1,b}+\Delta S_{2,b}
=\boxed{546.156~\mathrm{J/K}}=S_{\mathrm{gen},b}>0.
\end{align*}
% Use unrounded intermediate values: the two argon entropy terms nearly cancel.
