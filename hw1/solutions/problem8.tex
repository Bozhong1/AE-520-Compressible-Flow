\section*{Problem 8}

Steady ideal-gas flow; $V$: speed, $h$: mass-specific enthalpy,
$M$: molar mass, bars: molar quantities.
\[
q=0,\qquad w_{\mathrm{shaft}}=0,\qquad \Delta s=0,\qquad
\Delta z\simeq0,\qquad V_1\simeq0.
\]
\[
T_1=1200\ \mathrm{K},\qquad \frac{p_2}{p_1}=\frac{1}{50},\qquad
M=0.0280134\ \mathrm{kg/mol},\qquad
R_u=8.314462618\ \mathrm{J/(mol\,K)}.
\]

\partlabel{(a)}
% Notes L3, PDF p. 1 / printed p. 25, Eq. (75): CPG isentropic relation.
\textbf{Calorically perfect gas (CPG):} $\gamma=1.40$.
\[
\frac{T_2}{T_1}=\left(\frac{p_2}{p_1}\right)^{(\gamma-1)/\gamma}
\quad\Longrightarrow\quad
\boxed{T_{2,\mathrm{CPG}}=1200\left(\frac{1}{50}\right)^{0.4/1.4}
=392.429\ \mathrm{K}}.
\]

\partlabel{(b)}
% Notes L2, PDF p. 9 / printed p. 24, Eqs. (65)--(66), or
% L3, PDF p. 2 / printed p. 26, Eqs. (77)--(78): ideal-gas entropy.
% Multiply the notes' mass-specific relation by M to obtain the molar form.
\textbf{Thermally perfect gas (TPG):} $\bar c_p(T)=27.0+0.0040T$ in $\mathrm{J/(mol\,K)}$.
\begin{align*}
0&=\int_{T_1}^{T_2}\frac{\bar c_p(T)}{T}\,dT-R_u\ln\frac{p_2}{p_1},\\
\int_{1200}^{T_2}\left(\frac{27.0}{T}+0.0040\right)dT
&=R_u\ln\frac{1}{50},\\
27.0\ln\frac{T_2}{1200}+0.0040(T_2-1200)
&=8.314462618\ln\frac{1}{50}.
\end{align*}
MATLAB \texttt{fzero} on $[300,500]\ \mathrm{K}$:
\[
\boxed{T_{2,\mathrm{TPG}}=404.728\ \mathrm{K}},\qquad 0<T_2<T_1.
\]

\partlabel{(c)}
% Notes L2, PDF p. 9 / printed p. 24, Eq. (64): dh=cp(T)dT.
% Notes L3, PDF p. 15 / printed p. 39, Eq. (174): h+u^2/2=h_0.
% The homework uses V for the speed denoted u in the notes.
\[
h_1-h_2=\frac{1}{M}\int_{T_2}^{T_1}\bar c_p(T)\,dT,
\qquad
h_1+\frac{V_1^2}{2}=h_2+\frac{V_2^2}{2}
\quad\Longrightarrow\quad V_2\simeq\sqrt{2(h_1-h_2)}.
\]
% Notes L2, PDF p. 6 / printed p. 21, Eq. (40): cp-cv=R, gamma=cp/cv.
\textit{CPG:}
\begin{align*}
\bar c_p&=\frac{\gamma R_u}{\gamma-1}
=29.100619\ \mathrm{J/(mol\,K)},\\
h_1-h_2&=\frac{\bar c_p}{M}(T_1-T_2)
=\frac{29.100619}{0.0280134}(1200-392.429)
\simeq\boxed{838.913\ \mathrm{kJ/kg}},\\
V_{2,\mathrm{CPG}}&=\sqrt{2(838.913\times10^3)}
\simeq\boxed{1295.31\ \mathrm{m/s}}.
\end{align*}
\textit{TPG:}
\begin{align*}
h_1-h_2
&=\frac{1}{M}\int_{T_2}^{T_1}(27.0+0.0040T)\,dT\\
&=\frac{27.0(T_1-T_2)+0.0020(T_1^2-T_2^2)}{M}\\
&=\frac{27.0(1200-404.728)+0.0020(1200^2-404.728^2)}{0.0280134}\\
&\simeq\boxed{857.616\ \mathrm{kJ/kg}},\\
V_{2,\mathrm{TPG}}&=\sqrt{2(857.616\times10^3)}
\simeq\boxed{1309.67\ \mathrm{m/s}}.
\end{align*}

\partlabel{(d)}
% L3 notes PDF pp.1--2 (printed pp.25--26), Eqs. (72)--(78): CPG/TPG models.
\begin{align*}
\mathrm{CPG}:\quad &\bar c_p=\text{constant},\qquad \gamma=1.40,\\
\mathrm{TPG}:\quad &\bar c_p=27.0+0.0040T,\qquad
\gamma(T)=\frac{\bar c_p(T)}{\bar c_p(T)-R_u}.
\end{align*}
The heat-capacity model changes both $\int\bar c_p\,dT/T$ and $\int\bar c_p\,dT$:
\begin{gather*}
T_{2,\mathrm{TPG}}=404.728>392.429=T_{2,\mathrm{CPG}}\qquad[\mathrm K],\\
(h_1-h_2)_{\mathrm{TPG}}=857.616>838.913=(h_1-h_2)_{\mathrm{CPG}}
\qquad[\mathrm{kJ/kg}]\\
\Longrightarrow\quad V_{2,\mathrm{TPG}}>V_{2,\mathrm{CPG}}.
\end{gather*}
