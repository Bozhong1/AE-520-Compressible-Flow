\section*{Problem 2}

Steady, adiabatic, quasi-one-dimensional flow; isentropic except at the normal shock.
$1s$: before shock; $2s$: after shock; $e$: exit.
\[
\gamma=1.4,\qquad R=287~\mathrm{J/(kg\,K)},\qquad p=\rho RT,\qquad h=c_pT.
\]
\[
\frac{A_e}{A_t}=6,\qquad \frac{A_s}{A_t}=3,\qquad
p_{0,1}=800~\mathrm{kPa},\qquad T_{0,1}=600~\mathrm{K}.
\]

\partlabel{(a)}
$A_1^*=A_t$; $F(M)=A/A^*$ as in Problem 1.
% L12 notes PDF p.1 Eq. (228); p.2 Eq. (233); p.3 Eq. (240).
\[
F(M_{1s})=\frac{A_s}{A_1^*}=\frac{A_s}{A_t}=3,
\qquad M_{1s}>1
\quad\Longrightarrow\quad \boxed{M_{1s}=2.637416}.
\]
% Bisection: interval width < 1e-12; supersonic [1,10], subsonic [1e-7,1].

\partlabel{(b)}
% L9 notes PDF p.1, Eqs. (1)--(4).
\[
\rho_{1s}u_{1s}=\rho_{2s}u_{2s},\quad
p_{1s}+\rho_{1s}u_{1s}^2=p_{2s}+\rho_{2s}u_{2s}^2,\quad
h_{1s}+\frac{u_{1s}^2}{2}=h_{2s}+\frac{u_{2s}^2}{2}.
\]
% L9 notes PDF p.3, Eq. (21).
\[
M_{2s}=\left[\frac{1+\frac{\gamma-1}{2}M_{1s}^2}
{\gamma M_{1s}^2-\frac{\gamma-1}{2}}\right]^{1/2}
=\left[\frac{1+0.2(2.637416)^2}{1.4(2.637416)^2-0.2}\right]^{1/2}
=\boxed{0.500692}.
\]
Normal-shock stagnation-pressure ratio:
% L9 notes PDF p.4, Eq. (27); also L12 notes PDF p.5, Eq. (256).
\[
\Pi(M)\equiv\frac{p_{0,2}}{p_{0,1}}
=\left[\frac{(\gamma+1)M^2}{2+(\gamma-1)M^2}\right]^{\gamma/(\gamma-1)}
\left[\frac{\gamma+1}{2\gamma M^2-(\gamma-1)}\right]^{1/(\gamma-1)}.
\]
\[
\frac{p_{0,2}}{p_{0,1}}=\Pi(2.637416)=\boxed{0.446174},\qquad
p_{0,2}=800(0.446173792)=\boxed{356.939~\mathrm{kPa}}.
\]
$T_{0,2}=T_{0,1}=600~\mathrm{K}$.

\partlabel{(c)}
$A_2^*$: downstream sonic reference area.
% L12 notes PDF p.5, Eq. (251); p.4 Eq. (244) and p.6 Eq. (260).
\begin{gather*}
\dot m=\frac{p_0A^*}{\sqrt{T_0}}\sqrt{\frac{\gamma}{R}}
\left(\frac{2}{\gamma+1}\right)^{\frac{\gamma+1}{2(\gamma-1)}},\\
\dot m_1=\dot m_2,\quad T_{0,1}=T_{0,2}
\quad\Longrightarrow\quad p_{0,1}A_t=p_{0,2}A_2^*,\\
\frac{A_2^*}{A_t}=\frac{p_{0,1}}{p_{0,2}}
=\frac{800}{356.939034}=\boxed{2.241279}.
\end{gather*}

\partlabel{(d)}
% L12 notes PDF p.4, Eqs. (245)--(246).
\[
F(M_e)=\frac{A_e}{A_2^*}
=\frac{A_e/A_t}{A_2^*/A_t}
=\frac{6}{2.241279109}=2.67704275,
\quad 0<M_e<1
\quad\Longrightarrow\quad\boxed{M_e=0.222668}.
\]
% L10 notes PDF p.8, Eq. (101); L12 notes PDF p.2, Eq. (237).
\begin{align*}
T_{0,e}&=T_{0,1}=\boxed{600.000~\mathrm{K}},\qquad
p_{0,e}=p_{0,2}=\boxed{356.939~\mathrm{kPa}},\\
T_e&=\frac{T_{0,e}}{1+0.2M_e^2}
=\frac{600}{1+0.2(0.222667549)^2}=\boxed{594.109~\mathrm{K}},\\
p_e&=\frac{p_{0,e}}{(1+0.2M_e^2)^{3.5}}
=\frac{356.939034}{[1+0.2(0.222667549)^2]^{3.5}}
=\boxed{344.822~\mathrm{kPa}},\\
p_b&=p_e=\boxed{344.822~\mathrm{kPa}}.
\end{align*}

\partlabel{(e)}
$p_e=p_b=200~\mathrm{kPa}$.
% Derived from L12 notes PDF p.5 Eq. (251) and p.1 Eq. (223),
% using rho_e=p_b/(R T_e), u_e=M_e sqrt(gamma R T_e), and
% T_e=T01/[1+(gamma-1)M_e^2/2] (L10 PDF p.8 Eq. (101)).
\begin{gather*}
\frac{p_{0,1}A_t}{\sqrt{T_{0,1}}}\sqrt{\frac{\gamma}{R}}
\left(\frac{2}{\gamma+1}\right)^{\frac{\gamma+1}{2(\gamma-1)}}
=\rho_eu_eA_e=p_bA_eM_e\sqrt{\frac{\gamma}{RT_e}},\\
M_e\sqrt{1+\frac{\gamma-1}{2}M_e^2}
=\frac{p_{0,1}}{p_b}\frac{A_t}{A_e}
\left(\frac{2}{\gamma+1}\right)^{\frac{\gamma+1}{2(\gamma-1)}}
=\frac{800}{200}\frac{1}{6}\left(\frac{2}{2.4}\right)^3
=0.385802469.
\end{gather*}
$y=M_e^2>0$:
\[
0.2y^2+y-(0.385802469)^2=0,
\qquad y=0.144658338
\quad\Longrightarrow\quad \boxed{M_e=0.380340}.
\]
% L12 notes PDF p.2 Eq. (237), then L9 notes PDF p.4 Eq. (27).
\begin{align*}
p_{0,e}=p_{0,2}&=p_b(1+0.2M_e^2)^{3.5}
=200[1+0.2(0.380339767)^2]^{3.5}
=\boxed{220.995~\mathrm{kPa}},\\
\Pi(M_{1s})&=\frac{p_{0,2}}{p_{0,1}}
=\frac{220.995212}{800}=0.276244015
\quad\xrightarrow{M_{1s}>1}\quad M_{1s}=3.199936,\\
\frac{A_s}{A_t}&=F(M_{1s})=F(3.199935556)=\boxed{5.120645}.
\end{align*}
% Shock-recovery solve: bisection on [1,F_sup^{-1}(6)] = [1,3.367872...].
$1<A_s/A_t=5.120645<6,\quad 0<M_e=0.380340<1$.

\partlabel{(f)}
Shock outside the nozzle:
% L12 notes PDF p.4, Eqs. (247)--(250); L8 notes PDF p.5 Eq. (202).
\[
F(M_e)=6,\quad M_e>1
\quad\Longrightarrow\quad M_e=3.367872,\qquad
p_e=\frac{800}{[1+0.2(3.367872)^2]^{3.5}}=12.6726~\mathrm{kPa}.
\]
$u-a>0$ and $u+a>0$: disturbances travel downstream, so the back pressure
cannot propagate upstream through the supersonic exit.
The jet adjusts externally by compression waves or shocks if $p_e<p_b$,
or expansion waves if $p_e>p_b$; thus $p_e\ne p_b$ is possible.
